REPORT ZPURCHASE_ORDER_DETAILS_01.
TABLES : EKKO, EKPO, MAKT, LFA1, MSEG.

TYPES : BEGIN OF ST_PURCHASE,
        EBELN TYPE EKKO-EBELN,

        EBELP TYPE EKPO-EBELP,
        MATNR TYPE EKPO-MATNR,
        MENGE TYPE EKPO-MENGE,
        NETPR TYPE EKPO-NETPR,

        MAKTX TYPE MAKT-MAKTX,

        LIFNR TYPE LFA1-LIFNR,
        NAME1 TYPE LFA1-NAME1,

        MBLNR     TYPE MSEG-MBLNR,
        MJAHR     TYPE MSEG-MJAHR,
        MS_MENGE  TYPE MSEG-MENGE,
        REM_QTY   TYPE EKPO-MENGE,

  END OF ST_PURCHASE.

DATA : IT_PURCHASE TYPE TABLE OF ST_PURCHASE.
DATA : WA_PURCHASE TYPE ST_PURCHASE.

DATA: REMAINGING_QTY TYPE MSEG-MENGE.

DATA : IT_FCAT TYPE LVC_T_FCAT.
DATA : WA_FCAT TYPE LVC_S_FCAT.

SELECT-OPTIONS : S_EBELN FOR EKKO-EBELN.

START-OF-SELECTION.

SELECT
        A~EBELN

        B~EBELP
        B~MATNR
        B~MENGE
        B~NETPR

        C~MAKTX

        D~LIFNR
        D~NAME1

        F~MBLNR
        F~MJAHR
        F~MENGE AS MS_MENGE

 INTO TABLE IT_PURCHASE
 FROM EKKO AS A
 INNER JOIN EKPO AS B
 ON A~EBELN = B~EBELN

 LEFT OUTER JOIN MAKT AS C
 ON B~MATNR = C~MATNR

 INNER JOIN LFA1 AS D
 ON A~LIFNR = D~LIFNR

* LEFT OUTER JOIN MSEG AS F
* ON B~MENGE = F~MENGE

  LEFT OUTER JOIN MSEG AS F
  ON B~EBELN = F~EBELN
  AND B~EBELP = F~EBELP

 WHERE A~EBELN IN S_EBELN.



LOOP AT IT_PURCHASE INTO WA_PURCHASE.

  CLEAR REMAINGING_QTY.

  SELECT SUM( MENGE )
    INTO REMAINGING_QTY
    FROM MSEG
   WHERE EBELN = WA_PURCHASE-EBELN
     AND EBELP = WA_PURCHASE-EBELP.

  WA_PURCHASE-MS_MENGE = REMAINGING_QTY.

  WA_PURCHASE-REM_QTY = WA_PURCHASE-MENGE - REMAINGING_QTY.

  MODIFY IT_PURCHASE FROM WA_PURCHASE.

ENDLOOP.
